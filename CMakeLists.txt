set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

option(PT_BUILD_GUI "Build the raylib window (without it pdf-tools runs commands only)" ON)
option(PT_BUILD_PROGRAMS "Build pdf-tools and the tests (off: only the libraries, for the Android build)" ON)
option(PT_USE_PCRE2 "Regular expressions: search, redaction, finding personal data (PCRE2)" ON)
option(PT_USE_FONTCONFIG "Match PDF font names to installed fonts (fontconfig)" ON)
option(PT_USE_OPENSSL "Certificate signatures (OpenSSL libcrypto)" ON)
# -Wno-clobbered (below) is GCC's; clang (macOS) only needs to not warn about it

  add_compile_options(-Wno-unknown-warning-option)
endif()
set(PT_REPO ${CMAKE_CURRENT_SOURCE_DIR})
set(PT_ASSETS ${PT_REPO}/assets)
if(MSVC)

  add_compile_definitions(_CRT_SECURE_NO_WARNINGS _CRT_NONSTDC_NO_DEPRECATE)
else()
  set(PT_WARNINGS -Wall -Wextra -Wno-clobbered)
endif()

# Dependencies come from pkg-config (PKG_CONFIG_PATH for a prefix of your own); a path given on the command line
# (MUPDF_LIBRARY, PCRE2_LIBRARY, OPENSSL_ROOT_DIR) wins, which the cross builds (Android, macOS) use.
find_package(PkgConfig QUIET)

# Files compiled into the binaries as byte arrays: <symbol>[] and <symbol>_size (NUL-terminated).
function(pt_embed out symbol file)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${file})
  file(READ ${file} hex HEX)
  string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," bytes "${hex}")
  file(WRITE ${out}.tmp "const unsigned char ${symbol}[] = {${bytes}0x00};\nconst int ${symbol}_size = sizeof(${symbol}) - 1;\n")
  configure_file(${out}.tmp ${out} COPYONLY)
endfunction()

target_compile_options(ptcore PRIVATE ${PT_WARNINGS})
if(WIN32)
  # pt_compat.h's POSIX calls, the file association; <unistd.h>, <dirent.h> and <pthread.h> for MSVC
  target_sources(ptcore PRIVATE platform/windows/compat_win32.c platform/windows/desktop_win32.c)
  target_include_directories(ptcore PUBLIC $<BUILD_INTERFACE:${PT_REPO}/platform/windows/include>)
  target_link_libraries(ptcore PUBLIC advapi32 shell32)
else()
  target_link_libraries(ptcore PUBLIC m)
endif()
set(PT_PC_REQUIRES "")

# MuPDF 1.28
set(MUPDF_INCLUDE_DIR "" CACHE PATH "Directory containing mupdf/fitz.h")
set(MUPDF_LIBRARY "" CACHE FILEPATH "libmupdf.so, libmupdf.dylib or libmupdf.a")
set(MUPDF_THIRD_LIBRARY "" CACHE FILEPATH "libmupdf-third.a, for a static MuPDF")
if(MUPDF_LIBRARY)
  target_include_directories(ptcore PUBLIC ${MUPDF_INCLUDE_DIR})
  target_link_libraries(ptcore PUBLIC ${MUPDF_LIBRARY})
  if(MUPDF_THIRD_LIBRARY)
    target_link_libraries(ptcore PUBLIC ${MUPDF_THIRD_LIBRARY})
    # its HarfBuzz and Tesseract are C++ (MSVC's objects name their runtime themselves)
    if(APPLE)
      target_link_libraries(ptcore PUBLIC c++)
    elseif(NOT ANDROID AND NOT MSVC)
      target_link_libraries(ptcore PUBLIC stdc++ pthread)
    endif()
  endif()
else()
  pkg_check_modules(MUPDF REQUIRED IMPORTED_TARGET mupdf>=1.28)
  target_link_libraries(ptcore PUBLIC PkgConfig::MUPDF)
  string(APPEND PT_PC_REQUIRES " mupdf")
endif()
if(APPLE)
  # installed-font matching (fonts.c)
  target_link_libraries(ptcore PUBLIC "-framework CoreText" "-framework CoreFoundation")
endif()

if(PT_USE_PCRE2)
  if(PCRE2_LIBRARY)
    target_include_directories(ptcore PRIVATE ${PCRE2_INCLUDE_DIR})
    target_link_libraries(ptcore PUBLIC ${PCRE2_LIBRARY})
    target_compile_definitions(ptcore PUBLIC PT_USE_PCRE2=1)
    if(WIN32)
      # a static PCRE2: without it pcre2.h declares the functions as DLL imports
      target_compile_definitions(ptcore PRIVATE PCRE2_STATIC)
    endif()
  elseif(PkgConfig_FOUND)
    pkg_check_modules(PCRE2 IMPORTED_TARGET libpcre2-8)
    if(PCRE2_FOUND)
      target_link_libraries(ptcore PUBLIC PkgConfig::PCRE2)
      target_compile_definitions(ptcore PUBLIC PT_USE_PCRE2=1)
      string(APPEND PT_PC_REQUIRES " libpcre2-8")
    endif()
  endif()
  if(NOT PCRE2_LIBRARY AND NOT PCRE2_FOUND)
    message(STATUS "PCRE2 not found: regular expressions and finding personal data are off")
  endif()
endif()

if(PT_USE_FONTCONFIG AND PkgConfig_FOUND)
  pkg_check_modules(FONTCONFIG IMPORTED_TARGET fontconfig)
  if(FONTCONFIG_FOUND)
    target_link_libraries(ptcore PUBLIC PkgConfig::FONTCONFIG)
    target_compile_definitions(ptcore PUBLIC PT_USE_FONTCONFIG=1)
    string(APPEND PT_PC_REQUIRES " fontconfig")
  else()
    message(STATUS "fontconfig not found: text boxes fall back to standard and embedded fonts")
  endif()
endif()

if(PT_USE_OPENSSL)
  if(NOT OPENSSL_ROOT_DIR AND PkgConfig_FOUND)
    pkg_check_modules(LIBCRYPTO IMPORTED_TARGET libcrypto)
  endif()
  if(LIBCRYPTO_FOUND)
    target_link_libraries(ptcore PUBLIC PkgConfig::LIBCRYPTO)
    # a static libcrypto lists these only as private libraries
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
      target_link_libraries(ptcore PUBLIC dl pthread)
    endif()
    target_compile_definitions(ptcore PUBLIC PT_USE_OPENSSL=1)
    string(APPEND PT_PC_REQUIRES " libcrypto")
  else()
    find_package(OpenSSL COMPONENTS Crypto)
    if(OpenSSL_FOUND)
      target_link_libraries(ptcore PUBLIC OpenSSL::Crypto)
      if(WIN32)
        # what a static libcrypto calls in Windows
        target_link_libraries(ptcore PUBLIC crypt32 ws2_32 user32)
      endif()
      target_compile_definitions(ptcore PUBLIC PT_USE_OPENSSL=1)
    else()
      message(STATUS "OpenSSL not found: certificate signatures are off")
    endif()
  endif()
endif()

target_compile_definitions(ptcore PUBLIC PT_VERSION="${PROJECT_VERSION}")

  pt_embed(${PT_GEN}/ui_font.c pt_ui_font ${PT_ASSETS}/NotoSans-Regular.ttf)
  pt_embed(${PT_GEN}/ui_font_bold.c pt_ui_font_bold ${PT_ASSETS}/NotoSans-Bold.ttf)
  pt_embed(${PT_GEN}/man.c pt_asset_man ${PT_REPO}/man/pdf-tools.1)
  file(GLOB PT_ICONS ${PT_ASSETS}/icons/*.svg)
  set(PT_ICON_SOURCES)
  set(PT_ICON_TABLE "")
  set(PT_ICON_DECLS "")
  foreach(icon ${PT_ICONS})
    get_filename_component(name ${icon} NAME_WE)
    string(MAKE_C_IDENTIFIER ${name} id)
    pt_embed(${PT_GEN}/icon_${id}.c pt_icon_${id} ${icon})
    list(APPEND PT_ICON_SOURCES ${PT_GEN}/icon_${id}.c)
    string(APPEND PT_ICON_DECLS "extern const unsigned char pt_icon_${id}[];\nextern const int pt_icon_${id}_size;\n")
    string(APPEND PT_ICON_TABLE "  {\"${name}\", pt_icon_${id}, &pt_icon_${id}_size},\n")
  endforeach()
  file(WRITE ${PT_GEN}/icons.c.tmp "#include \"ui_icons.h\"\n${PT_ICON_DECLS}const pt_icon_data pt_icons[] = {\n${PT_ICON_TABLE}  {0, 0, 0}\n};\n")
  configure_file(${PT_GEN}/icons.c.tmp ${PT_GEN}/icons.c COPYONLY)
  file(GLOB PT_UI_SOURCES CONFIGURE_DEPENDS src/ui/*.c)
  add_executable(pdf-tools src/main.c ${PT_UI_SOURCES} ${PT_MANIFEST}

  target_compile_definitions(pdf-tools PRIVATE PT_GUI=1)
  target_include_directories(pdf-tools PRIVATE src/ui thirdparty/raygui/src)
  target_link_libraries(pdf-tools PRIVATE ptcli raylib)
  if(APPLE)
    target_link_libraries(pdf-tools PRIVATE "-framework Cocoa" "-framework IOKit" "-framework OpenGL"
      "-framework CoreVideo" "-framework CoreAudio" "-framework AudioToolbox")
  endif()
  if(WIN32)
    # other windows, printing, programs, file dialogs and the Recycle Bin (platform/windows/pt_win32.h)
    target_sources(pdf-tools PRIVATE platform/windows/system_win32.c platform/windows/dialogs_win32.c)
    target_include_directories(pdf-tools PRIVATE platform/windows)
    target_link_libraries(pdf-tools PRIVATE ws2_32 ole32 uuid user32 gdi32 winspool winmm opengl32)
  endif()
  if(MSVC)
    target_compile_options(pdf-tools PRIVATE ${PT_WARNINGS})
  else()
    target_compile_options(pdf-tools PRIVATE ${PT_WARNINGS} -Wno-unused-parameter)
  endif()
endif()
