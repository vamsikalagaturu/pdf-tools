cmake_minimum_required(VERSION 3.20)
project(pdf_tools_c VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(PT_BUILD_GUI "Build raylib desktop editor" ON)
option(PT_BUILD_TESTS "Build native unit tests" ON)

# MuPDF is distributed in multiple forms; caller can set explicit include and library.
set(MUPDF_INCLUDE_DIR "" CACHE PATH "Directory containing mupdf/fitz.h")
set(MUPDF_LIBRARY "" CACHE FILEPATH "Path to libmupdf.so, libmupdf.dylib or libmupdf.a")
if(NOT MUPDF_INCLUDE_DIR)
  find_path(MUPDF_INCLUDE_DIR NAMES mupdf/fitz.h REQUIRED)
endif()
if(NOT MUPDF_LIBRARY)
  find_library(MUPDF_LIBRARY NAMES mupdf REQUIRED)
endif()
add_library(ptcore STATIC src/pdf.c src/ranges.c)
target_include_directories(ptcore PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include ${MUPDF_INCLUDE_DIR})
target_link_libraries(ptcore PUBLIC ${MUPDF_LIBRARY} m)
if(MSVC)
  target_compile_options(ptcore PRIVATE /W4)
else()
  target_compile_options(ptcore PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(pdf-tools-c src/cli.c)
target_link_libraries(pdf-tools-c PRIVATE ptcore)

if(PT_BUILD_GUI)
  find_package(raylib 5 QUIET CONFIG)
  if(raylib_FOUND)
    add_executable(pdf-tools-gui src/gui.c)
    target_link_libraries(pdf-tools-gui PRIVATE ptcore raylib)
  else()
    message(STATUS "raylib not found; CLI builds, GUI skipped. Install raylib and reconfigure.")
  endif()
endif()

if(PT_BUILD_TESTS)
  enable_testing()
  add_executable(pt-test-ranges tests/test_ranges.c src/ranges.c)
  target_include_directories(pt-test-ranges PRIVATE include)
  add_test(NAME page_ranges COMMAND pt-test-ranges)
endif()
if(PT_BUILD_TESTS)
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND)
    add_test(NAME cli_integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_e2e.py $<TARGET_FILE:pdf-tools-c>)
  endif()
endif()
