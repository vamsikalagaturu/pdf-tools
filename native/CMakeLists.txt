if(NOT MUPDF_LIBRARY)
  find_library(MUPDF_LIBRARY NAMES mupdf REQUIRED)
endif()

# Files compiled into the binaries as byte arrays: <symbol>[] and <symbol>_size (NUL-terminated).
function(pt_embed out symbol file)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${file})
  file(READ ${file} hex HEX)
  string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," bytes "${hex}")
  file(WRITE ${out}.tmp "const unsigned char ${symbol}[] = {${bytes}0x00};\nconst int ${symbol}_size = sizeof(${symbol}) - 1;\n")
  configure_file(${out}.tmp ${out} COPYONLY)
endfunction()

add_library(ptcore STATIC
  src/core/util.c src/core/ranges.c src/core/text.c src/core/forms.c src/core/fonts.c src/core/edits.c
  src/core/synctex.c src/core/selection.c src/core/edit_list.c src/core/config.c src/core/signatures.c
  src/core/desktop.c src/core/pii.c src/core/stamps.c src/core/log.c ${PT_GEN}/logo_svg.c ${PT_GEN}/desktop.c)
target_include_directories(ptcore PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include ${MUPDF_INCLUDE_DIR})
target_link_libraries(ptcore PUBLIC ${MUPDF_LIBRARY} m)
# a static MuPDF build (macOS, Android) keeps its third-party libraries in a second archive
set(MUPDF_THIRD_LIBRARY "" CACHE FILEPATH "libmupdf-third.a, for a static MuPDF")
if(MUPDF_THIRD_LIBRARY)
  target_link_libraries(ptcore PUBLIC ${MUPDF_THIRD_LIBRARY})
  # its HarfBuzz is C++
  if(APPLE)
    target_link_libraries(ptcore PUBLIC c++)
  endif()
endif()
if(APPLE)
  # installed-font matching (fonts.c)
  target_link_libraries(ptcore PUBLIC "-framework CoreText" "-framework CoreFoundation")
endif()
if(MSVC)
  target_compile_options(ptcore PRIVATE /W4)
else()
  target_compile_options(ptcore PRIVATE -Wall -Wextra -Wno-clobbered)
endif()

option(PT_USE_PCRE2 "Regular-expression redaction (PCRE2)" ON)
if(PT_USE_PCRE2)
  find_path(PCRE2_INCLUDE_DIR NAMES pcre2.h)
  find_library(PCRE2_LIBRARY NAMES pcre2-8)
  if(PCRE2_INCLUDE_DIR AND PCRE2_LIBRARY)
    target_compile_definitions(ptcore PUBLIC PT_USE_PCRE2=1)
    target_include_directories(ptcore PRIVATE ${PCRE2_INCLUDE_DIR})
    target_link_libraries(ptcore PUBLIC ${PCRE2_LIBRARY})
  else()
    message(STATUS "PCRE2 not found; --regex disabled")
  endif()
endif()
option(PT_USE_FONTCONFIG "Match PDF font names to installed fonts (fontconfig)" ON)
if(PT_USE_FONTCONFIG)
  find_path(FONTCONFIG_INCLUDE_DIR NAMES fontconfig/fontconfig.h)
  find_library(FONTCONFIG_LIBRARY NAMES fontconfig)
  if(FONTCONFIG_INCLUDE_DIR AND FONTCONFIG_LIBRARY)
    target_compile_definitions(ptcore PUBLIC PT_USE_FONTCONFIG=1)
    target_include_directories(ptcore PRIVATE ${FONTCONFIG_INCLUDE_DIR})
    target_link_libraries(ptcore PUBLIC ${FONTCONFIG_LIBRARY})
  else()
    message(STATUS "fontconfig not found; text boxes fall back to standard and embedded fonts")
  endif()
endif()
